-- 1) Sicherheit einschalten alter table public.lernraum_sync enable row level security; -- 2) Alle alten Regeln entfernen (falls welche zu offen waren) do $$ declare p record; begin for p in select policyname from pg_policies where schemaname = 'public' and tablename = 'lernraum_sync' loop execute format('drop policy %I on public.lernraum_sync', p.policyname); end loop; end $$; -- 3) Neue Regeln: Jeder sieht und ändert NUR seine eigenen Daten create policy "eigene_daten_lesen" on public.lernraum_sync for select to authenticated using (auth.uid()::text = user_id::text); create policy "eigene_daten_anlegen" on public.lernraum_sync for insert to authenticated with check (auth.uid()::text = user_id::text); create policy "eigene_daten_aendern" on public.lernraum_sync for update to authenticated using (auth.uid()::text = user_id::text) with check (auth.uid()::text = user_id::text); create policy "eigene_daten_loeschen" on public.lernraum_sync for delete to authenticated using (auth.uid()::text = user_id::text); -- 4) Kontrolle: zeigt, ob RLS an ist und welche Regeln gelten select c.relrowsecurity as rls_aktiv, p.policyname, p.cmd from pg_class c left join pg_policies p on p.tablename = c.relname where c.relname = 'lernraum_sync';
